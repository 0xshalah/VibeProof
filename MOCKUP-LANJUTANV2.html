template AuditProof() {
  signal input  bytecodeHash;   // privat
  signal input  auditScore;     // privat
  signal input  auditorPk;      // privat
  signal output minScoreMet;    // publik

  // membuktikan: score > 90 &&
  // hash == keccak256(source) &&
  // signature auditor valid
  minScoreMet <-- (auditScore > 90) ? 1 : 0;
  (auditScore - 90) * inv === minScoreMet;
}
